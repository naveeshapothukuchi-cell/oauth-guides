import { NextResponse } from 'next/server'

const SYSTEM_PROMPT = `You are a research agent that creates clear OAuth setup guides for developers and AI agent builders.

export async function POST(request: Request) {
  try {
    const { app } = await request.json()

    if (!app) {
      return NextResponse.json({ error: 'App name or URL is required' }, { status: 400 })
    }

    // TODO: Replace this with a real LLM call later
    const guide = `Guide for: ${app}\n\n(LLM response will appear here once connected.)`

    return NextResponse.json({ guide })
  } catch (error) {
    return NextResponse.json({ error: 'Failed to generate guide' }, { status: 500 })
  }
}
